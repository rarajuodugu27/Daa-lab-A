        "# Coin Change Problem using Dynamic Programming\n",
        "\n",
        "def coin_change(coins, amount):\n",
        "\n",
        "    # dp[i] = minimum coins required to make amount i\n",
        "    dp = [float('inf')] * (amount + 1)\n",
        "\n",
        "    # Base case: 0 coins are needed to make amount 0\n",
        "    dp[0] = 0\n",
        "\n",
        "    # Build the DP table\n",
        "    for i in range(1, amount + 1):\n",
        "        for coin in coins:\n",
        "            if coin <= i:\n",
        "                dp[i] = min(dp[i], dp[i - coin] + 1)\n",
        "\n",
        "    # If amount cannot be formed\n",
        "    if dp[amount] == float('inf'):\n",
        "        return -1\n",
        "\n",
        "    return dp[amount]\n",
        "\n",
        "\n",
        "# Taking input from user\n",
        "n = int(input(\"Enter number of coins: \"))\n",
        "\n",
        "coins = []\n",
        "print(\"Enter the coin values:\")\n",
        "\n",
        "for i in range(n):\n",
        "    coin = int(input(f\"Coin {i + 1}: \"))\n",
        "    coins.append(coin)\n",
        "\n",
        "amount = int(input(\"Enter the amount: \"))\n",
        "\n",
        "# Calculate answer\n",
        "answer = coin_change(coins, amount)\n",
        "\n",
        "print(\"Minimum number of coins:\", answer)\n",
        "\n",
        "print(\"Time Complexity  : O(n * amount)\")\n",
